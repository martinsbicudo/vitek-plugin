Vitek Socket Example

WebSocket-only example. Connect to /ws (root) or /ws/chat.

Live demo

Using generated services

import { connect, connectChat } from './socket.services.js';

// Connect to root ws
const ws = connect();
ws.onmessage = (e) => console.log(e.data);
ws.send('Hello');

// Connect to chat
const chat = connectChat();
chat.onmessage = (e) => console.log(e.data);
